Integrasi Live Google Sheets

Sambungkan sistem imbasan ini secara terus ke Google Sheet sekolah anda

Tutup

Sediakan Web App melalui Google Apps Script di Google Sheet anda dan masukkan URL penyebaran (deployment URL) di sini.

Kod Google Apps Script Multi-Device (Code.gs) 
// KOD GOOGLE APPS SCRIPT UNTUK SIMPAN & AMBIL DATA LIVE
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.id || ("ATT_" + new Date().getTime()),
    data.studentId,
    data.studentName,
    data.studentClass,
    data.date,
    data.scanTime,
    data.status
  ]);
  return ContentService.createTextOutput(JSON.stringify({"result":"success"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var records = [];
  
  // Mula dari indeks 1 untuk melangkau header/tajuk kolum
  for (var i = 1; i < rows.length; i++) {
    if(rows[i][0]) {
      records.push({
        id: rows[i][0],
        studentId: rows[i][1],
        studentName: rows[i][2],
        studentClass: rows[i][3],
        date: rows[i][4] ? String(rows[i][4]) : "",
        scanTime: rows[i][5] ? String(rows[i][5]) : "",
        status: rows[i][6] || "Hadir",
        syncStatus: "Synced"
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify(records)).setMimeType(ContentService.MimeType.JSON);
}
