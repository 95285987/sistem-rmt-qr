Kod Google Apps Script (.gs) Terkini

// ==========================================
// KOD GOOGLE APPS SCRIPT (Code.gs)
// SISTEM KEHADIRAN RMT QR CODE
// ==========================================

// 1. Fungsi GET: Membaca rekod kehadiran dari Google Sheet secara Live
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var logs = [];
  
  // Membaca semua baris data (mengabaikan tajuk kolum di baris 1)
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0] !== "" && rows[i][0] !== undefined) {
      logs.push({
        nama: String(rows[i][0]),
        kelas: String(rows[i][1]),
        tarikh: String(rows[i][2]),
        masa: String(rows[i][3]),
        status: String(rows[i][4])
      });
    }
  }
  
  // Susun rekod supaya imbasan terkini berada di atas
  logs.reverse();
  
  return ContentService.createTextOutput(JSON.stringify({
    "result": "success", 
    "data": logs
  })).setMimeType(ContentService.MimeType.JSON);
}

// 2. Fungsi POST: Menyimpan imbasan baru ke Google Sheet
function doPost(e) {
  var lock = LockService.getScriptLock();
  // Kunci 10 saat elak isu data bertindih jika ramai murid imbas serentak
  lock.tryLock(10000); 
  
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Tambah baris baru di bawah tajuk kolum
    sheet.appendRow([
      data.nama, 
      data.kelas, 
      data.tarikh, 
      data.masa, 
      data.status
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({
      "result": "success"
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      "result": "error", 
      "message": err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
    
  } finally {
    lock.releaseLock();
  }
}
